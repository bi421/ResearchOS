#include "quant_engine.h"
#include <iostream>

namespace cpp_quant {

QuantEngine::QuantEngine() {}
QuantEngine::~QuantEngine() {}

bool QuantEngine::loadData(const std::vector<std::string>& csv_files) {
    raw_data_ = DataLoader::mergeFiles(csv_files);
    if (raw_data_.empty()) {
        data_loaded_ = false;
        return false;
    }
    aggregateData();
    data_loaded_ = true;
    return true;
}

bool QuantEngine::loadDataFromVectors(
    const std::vector<int64_t>& timestamps,
    const std::vector<double>& opens,
    const std::vector<double>& highs,
    const std::vector<double>& lows,
    const std::vector<double>& closes,
    const std::vector<double>& volumes) {

    raw_data_.resize(timestamps.size());
    for (size_t i = 0; i < timestamps.size(); ++i) {
        raw_data_[i].timestamp = timestamps[i];
        raw_data_[i].open = opens[i];
        raw_data_[i].high = highs[i];
        raw_data_[i].low = lows[i];
        raw_data_[i].close = closes[i];
        raw_data_[i].volume = volumes[i];
    }
    aggregateData();
    data_loaded_ = true;
    return true;
}

void QuantEngine::setTimeframe(int minutes) {
    if (minutes == timeframe_minutes_) return;
    timeframe_minutes_ = minutes;
    if (data_loaded_) aggregateData();
}

void QuantEngine::aggregateData() {
    if (raw_data_.empty()) return;
    if (timeframe_minutes_ == 1) {
        data_ = raw_data_;
    } else {
        data_ = DataLoader::aggregate(raw_data_, timeframe_minutes_);
    }
}

BacktestResult QuantEngine::runSMA(int short_period, int long_period) {
    if (!data_loaded_ || data_.empty()) return BacktestResult();
    BacktestEngine engine(data_, 10000.0, 0.0001);
    return engine.runSMA(short_period, long_period);
}

BacktestResult QuantEngine::runRSI(int period, double oversold, double overbought) {
    if (!data_loaded_ || data_.empty()) return BacktestResult();
    BacktestEngine engine(data_, 10000.0, 0.0001);
    return engine.runRSI(period, oversold, overbought);
}

BacktestResult QuantEngine::runMACD(int fast, int slow, int signal, int sma_filter) {
    if (!data_loaded_ || data_.empty()) return BacktestResult();
    BacktestEngine engine(data_, 10000.0, 0.0001);
    return engine.runMACD(fast, slow, signal, sma_filter);
}

std::map<std::string, BacktestResult> QuantEngine::runAllStrategies() {
    std::map<std::string, BacktestResult> results;
    if (!data_loaded_ || data_.empty()) return results;
    results["SMA_10_30"] = runSMA(10, 30);
    results["SMA_20_50"] = runSMA(20, 50);
    results["RSI_14"] = runRSI(14, 30, 70);
    results["MACD_12_26_9"] = runMACD(12, 26, 9, 200);
    return results;
}

double QuantEngine::monteCarloPValue(const BacktestResult& result, int num_simulations) {
    if (result.trades.empty()) return 1.0;
    return MonteCarlo::pValue(result.trades, num_simulations);
}

std::map<std::string, double> QuantEngine::optimizeSMA(int short_min, int short_max, int long_min, int long_max) {
    if (!data_loaded_ || data_.empty()) return {};
    return Optimizer::optimizeSMA(data_, short_min, short_max, 5, long_min, long_max, 10);
}

std::string QuantEngine::getDataInfo() const {
    if (!data_loaded_) return "No data loaded.";
    std::string info = "Data: " + std::to_string(data_.size()) + " candles, ";
    if (!data_.empty()) {
        info += "from " + std::to_string(data_.front().timestamp) + " to " + std::to_string(data_.back().timestamp);
    }
    return info;
}

} // namespace cpp_quant
