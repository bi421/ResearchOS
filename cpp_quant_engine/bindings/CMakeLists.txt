if(NOT TARGET quant_engine)
    message(FATAL_ERROR "bindings/ must be configured from the root")
endif()
find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_VARIABLE nanobind_DIR OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
find_package(nanobind CONFIG REQUIRED PATHS "${nanobind_DIR}" NO_DEFAULT_PATH)
nanobind_add_module(cpp_quant_backend python_bindings.cpp)
target_link_libraries(cpp_quant_backend PRIVATE quant_engine)
target_include_directories(cpp_quant_backend PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../include ${CMAKE_CURRENT_SOURCE_DIR}/../python)
set_target_properties(cpp_quant_backend PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../python/cpp_quant_engine")
